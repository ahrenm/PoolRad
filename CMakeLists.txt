cmake_minimum_required(VERSION 3.25)
project(PoolRad VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
# Match the validated release optimization level, retaining custom user flags.
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND CMAKE_CXX_FLAGS_RELEASE STREQUAL "-O3 -DNDEBUG")
    set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG" CACHE STRING "Release compiler flags" FORCE)
endif()

include(cmake/SDL3.cmake)

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/config/source_sets.json" source_manifest)
foreach(block core tools modes application bootstrap)
    string(JSON count LENGTH "${source_manifest}" "${block}")
    math(EXPR last "${count} - 1")
    foreach(index RANGE 0 ${last})
        string(JSON source GET "${source_manifest}" "${block}" ${index})
        if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
            message(FATAL_ERROR "Missing ${block} source: ${source}")
        endif()
        list(APPEND poolrad_${block}_sources "${source}")
    endforeach()
endforeach()

function(poolrad_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endfunction()

add_library(gbengine_core STATIC ${poolrad_core_sources})
target_include_directories(gbengine_core PUBLIC include PRIVATE third_party)
if(WIN32)
    target_link_libraries(gbengine_core PUBLIC advapi32 windowscodecs ole32 uuid)
endif()
add_library(gbengine_tools STATIC ${poolrad_tools_sources})
target_link_libraries(gbengine_tools PUBLIC gbengine_core)
add_library(gbengine_modes STATIC ${poolrad_modes_sources})
target_link_libraries(gbengine_modes PUBLIC gbengine_core)
add_library(gbengine_application STATIC ${poolrad_application_sources})
target_link_libraries(gbengine_application PUBLIC gbengine_modes gbengine_tools SDL3::SDL3-shared)
if(WIN32)
    target_link_libraries(gbengine_application PRIVATE user32 gdi32)
endif()
add_executable(poolrad ${poolrad_bootstrap_sources})
target_link_libraries(poolrad PRIVATE gbengine_application)
if(WIN32)
    set_target_properties(poolrad PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
foreach(target gbengine_core gbengine_tools gbengine_modes gbengine_application poolrad)
    poolrad_warnings(${target})
endforeach()

install(TARGETS poolrad RUNTIME DESTINATION .)
install(FILES README.md DESTINATION .)
install(FILES third_party/tomlplusplus/LICENSE DESTINATION licenses RENAME tomlplusplus-LICENSE)
install(FILES third_party/SDL3-LICENSE.txt DESTINATION licenses)
if(WIN32)
    add_custom_command(TARGET poolrad POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:SDL3::SDL3-shared> $<TARGET_FILE_DIR:poolrad>)
    install(FILES $<TARGET_FILE:SDL3::SDL3-shared> DESTINATION .)
    # Runtime DLLs come from the selected compiler, never the developer workspace.
    if(MINGW)
        get_filename_component(compiler_directory "${CMAKE_CXX_COMPILER}" DIRECTORY)
        foreach(name libgcc_s_seh-1.dll libstdc++-6.dll libwinpthread-1.dll libmcfgthread-2.dll)
            if(EXISTS "${compiler_directory}/${name}")
                add_custom_command(TARGET poolrad POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    "${compiler_directory}/${name}" $<TARGET_FILE_DIR:poolrad>)
                install(FILES "${compiler_directory}/${name}" DESTINATION .)
            endif()
        endforeach()
    endif()
    find_program(POOLRAD_POWERSHELL NAMES powershell pwsh REQUIRED)
    configure_file(cmake/InstallShortcut.cmake.in InstallShortcut.cmake @ONLY)
    install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/InstallShortcut.cmake")
endif()
